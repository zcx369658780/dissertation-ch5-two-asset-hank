class Stop(RuntimeError):
    def __init__(self, code):
        self.code = code
        super().__init__()  # Never attach foreign exception data.

def require(ok, code):
    if not ok:
        raise Stop(code)

def digest(raw):
    return hashlib.sha256(raw).hexdigest()

def blob(raw):
    return hashlib.sha1(b'blob '+str(len(raw)).encode('utf-8')+b'\0'+raw).hexdigest()

def exact_path(text):
    require(type(text) is str and '\0' not in text, 'PATH_TYPE')
    require(ntpath.isabs(text) and not text.startswith('\\\\'), 'PATH_ABSOLUTE')
    require('/' not in text and ':' not in text[2:], 'PATH_ALIAS')
    require(not any(x in ('.','..') for x in text.split('\\')), 'PATH_TRAVERSAL')
    return Path(text)

def chain(path, missing_leaf=False):
    p = exact_path(str(path))
    nodes = (p,*p.parents) if not missing_leaf else tuple(p.parents)
    for x in nodes:
        s = x.lstat()
        require(not (s.st_file_attributes & 0x400), 'PATH_REPARSE')
    require(missing_leaf or p.resolve() == p, 'PATH_RESOLUTION')
    return p

def small_error(exc):

    if type(exc) is Stop:

          'start':exc.start,'end':exc.end,'reason':exc.reason,
          'input_bytes':len(exc.object),'input_sha256':digest(exc.object)}
    if (type(exc).__module__=='ch5_two_asset_hank.multi_province.mainline_household_runtime'
            and type(exc).__name__=='ScienceFailure'):

class Writer:
    """Single trusted writer; all temporary bytes remain charged on failure."""
    def __init__(self, root):
        self.root = exact_path(root)
        chain(self.root,missing_leaf=True)
        require(not self.root.exists(), 'OUTPUT_ALREADY_EXISTS')
        self.root.mkdir()  # Bootstrap, before hook; one fresh authorized directory.
        self.lock = threading.RLock()
        self.ticket = threading.local()
        self.used = self.pending = 0
        self.log_bytes = 0
        self.records = []

    def _target(self,name):
        p = exact_path(str(self.root/name))
        require(p.is_relative_to(self.root), 'OUTPUT_ESCAPE')
        chain(p,missing_leaf=True)
        require(not p.exists(), 'OUTPUT_EXISTS')
        return p

    def mkdir(self,name):
        p = self._target(name)
        self.ticket.event = ('mkdir',str(p))
        try:
            p.mkdir()
        finally:
            self.ticket.event = None

    def write(self,name,raw,limit,terminal=False):
        require(type(raw) is bytes and len(raw)<=limit, 'WRITE_SINGLE_CAP')
        with self.lock:
            p = self._target(name)
            require(self.used+self.pending+len(raw) <= (CAP if terminal else CAP-RESERVE),
                    'WRITE_TOTAL_CAP')
            tmp = self.root/('.txn_'+uuid.uuid4().hex)
            self.used += len(raw)  # Reserve before creating any artifact.
            self.ticket.event = ('open',str(tmp))
            fd = None
            try:
                fd = os.open(tmp,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_BINARY,0o600)
                self.ticket.event = None
                view = memoryview(raw)
                while view:
                    n = os.write(fd,view)
                    require(n>0,'WRITE_SHORT')
                    view = view[n:]
                os.fsync(fd)
                os.close(fd); fd = None
                self.ticket.event = ('rename',str(tmp),str(p))
                # Windows rename is atomic and refuses an existing destination.
                if getattr(self,'inject_rename_fault',False):
                    raise OSError('CONTROLLED_ENGINEERING_RENAME_FAULT')
                os.rename(tmp,p)
                self.records.append({'name':name,'bytes':len(raw),'sha256':digest(raw)})
            finally:
                self.ticket.event = None
                if fd is not None:
                    os.close(fd)
                # Do not erase partial/new evidence or release uncertain bytes.

    def json(self,name,value,model=False,terminal=False):
        raw = (json.dumps(value,ensure_ascii=False,sort_keys=True,indent=2,
                          allow_nan=False)+'\n').encode('utf-8',errors='strict')

    def log_append(self,raw):
        with self.lock:
            require(self.log_bytes+len(raw)<=LOG_MAX,'LOG_CAP')
            require(self.used+self.pending+len(raw)<=CAP-RESERVE,'WRITE_TOTAL_CAP')
            self.pending += len(raw)
            self.log_bytes += len(raw)

    def log_seal(self,name,raw):
        with self.lock:
            # Transfer, never double charge, the pending reservation.
            before = self.used
            self.pending -= len(raw)
            try:
                self.write(name,raw,LOG_MAX)
            except BaseException:
                if self.used==before:
                    self.pending += len(raw)
                # Otherwise the writer already charged a retained transaction.
                raise

class Log(io.TextIOBase):
    def __init__(self,writer,name):
        self.writer,self.name,self.parts = writer,name,[]
    @property
    def encoding(self):
        return 'utf-8'
    def writable(self):
        return True
    def flush(self):
        return None
    def write(self,text):
        require(type(text) is str and len(text)<=LOG_MAX,'LOG_INPUT_CAP')
        raw = text.encode('utf-8',errors='strict')
        self.writer.log_append(raw)
        self.parts.append(raw)
        return len(text)
    def seal(self):
        self.writer.log_seal(self.name,b''.join(self.parts))

class WindowsJob:
    """Kill-on-close ownership of a trusted read-only Git child/descendants.

    Assignment failure stops before identity use. Trusted installation and no
    hostile pre-assignment escape are explicit assumptions, not OS sandbox proof.
    """
    def __init__(self,child):
        class Basic(ctypes.Structure):
            _fields_=[('user',ctypes.c_longlong),('job',ctypes.c_longlong),
              ('flags',wintypes.DWORD),('minws',ctypes.c_size_t),('maxws',ctypes.c_size_t),
              ('active',wintypes.DWORD),('affinity',ctypes.c_size_t),
              ('priority',wintypes.DWORD),('scheduling',wintypes.DWORD)]
        class IO(ctypes.Structure):
            _fields_=[('n'+str(i),ctypes.c_ulonglong) for i in range(6)]
        class Extended(ctypes.Structure):
            _fields_=[('basic',Basic),('io',IO),('process_mem',ctypes.c_size_t),
              ('job_mem',ctypes.c_size_t),('peak_process',ctypes.c_size_t),('peak_job',ctypes.c_size_t)]
        self.k = ctypes.WinDLL('kernel32',use_last_error=True)
        self.k.CreateJobObjectW.restype = wintypes.HANDLE
        self.k.CreateJobObjectW.argtypes = [ctypes.c_void_p,wintypes.LPCWSTR]
        self.k.SetInformationJobObject.argtypes = [wintypes.HANDLE,ctypes.c_int,ctypes.c_void_p,wintypes.DWORD]
        self.k.AssignProcessToJobObject.argtypes = [wintypes.HANDLE,wintypes.HANDLE]
        self.k.CloseHandle.argtypes = [wintypes.HANDLE]
        self.handle = self.k.CreateJobObjectW(None,None)
        require(bool(self.handle),'CHILD_JOB_CREATE')
        info = Extended(); info.basic.flags = 0x2000
        try:
            require(bool(self.k.SetInformationJobObject(self.handle,9,ctypes.byref(info),ctypes.sizeof(info))),
                    'CHILD_JOB_LIMIT')
            require(bool(self.k.AssignProcessToJobObject(self.handle,wintypes.HANDLE(int(child._handle)))),
                    'CHILD_JOB_ASSIGN')
        except BaseException:
            self.close()
            raise
    def close(self):
        if self.handle:
            self.k.CloseHandle(self.handle)
            self.handle = None

def bounded_child(plan,hook,proof):
    """Only spawn site. Two readers request at most remaining+one probe byte."""
    child = job = None
    state = {'fault':None,'stdout':bytearray(),'stderr':bytearray()}
    readers = []
    lock = threading.Lock()
    start = time.monotonic()
    def read_pipe(pipe,name,limit):
        try:
            while True:
                with lock:
                    allowance = limit-len(state[name])
                piece = pipe.read(min(8192,allowance+1))
                if not piece:
                    return
                with lock:
                    if len(piece)>allowance:
                        state['fault']='CHILD_'+name.upper()+'_CAP'
                        return  # The overflow probe is never appended.
                    state[name].extend(piece)
        except BaseException:
            state['fault']='CHILD_PIPE_READ'
    try:
        require(getattr(hook.spawn,'plan',None) is None,'GIT_CAPABILITY_REENTRY')
        hook.spawn.plan = plan
        try:
            child = subprocess.Popen(plan['argv'],executable=plan['exe'],shell=False,
              cwd=plan['cwd'],env=plan['env'],stdin=subprocess.DEVNULL,
              stdout=subprocess.PIPE,stderr=subprocess.PIPE,bufsize=0,close_fds=True,
              creationflags=subprocess.CREATE_NO_WINDOW)
        finally:
            hook.spawn.plan = None
        require(plan['audit_consumed'],'GIT_AUDIT_NOT_CONSUMED')
        job = WindowsJob(child)
        for name,pipe,limit in (('stdout',child.stdout,plan['out_cap']),
                                ('stderr',child.stderr,4096)):
            t = threading.Thread(target=read_pipe,args=(pipe,name,limit),daemon=True)
            readers.append(t); t.start()
        while child.poll() is None or any(t.is_alive() for t in readers):
            require(state['fault'] is None,state['fault'] or 'CHILD_PIPE_READ')
            require(time.monotonic()-start<CHILD_TIMEOUT,'CHILD_TIMEOUT')
            time.sleep(0.01)
        require(state['fault'] is None,state['fault'] or 'CHILD_PIPE_READ')
        require(child.wait(timeout=REAP_TIMEOUT)==0,'CHILD_EXIT_NONZERO')
        return bytes(state['stdout']),bytes(state['stderr'])
    finally:
        proof.update({'nonce':plan['nonce'],'audit_consumed':plan['audit_consumed'],
          'stdout_bytes':len(state['stdout']),'stderr_bytes':len(state['stderr']),
          'stdout_sha256':digest(state['stdout']),'stderr_sha256':digest(state['stderr']),
          'fault':state['fault'],'elapsed_seconds':time.monotonic()-start,
          'timeout':time.monotonic()-start>=CHILD_TIMEOUT,'job_assigned':job is not None,
          'streams_complete':state['fault'] is None and child is not None and child.poll() is not None,
          'exit_code':None,'reaped':False,'readers_joined':False})
        if job is not None:
            job.close()  # Kills owned remaining descendants on any exit path.
        if child is not None:
            if child.poll() is None:
                child.kill()
            try:
                child.wait(timeout=REAP_TIMEOUT)
            except BaseException:
                raise Stop('CHILD_REAP_FAILED')
            proof['exit_code']=child.returncode
            proof['reaped']=True
            for pipe in (child.stdout,child.stderr):
                if pipe is not None:
                    pipe.close()
            for t in readers:
                t.join(timeout=REAP_TIMEOUT)
            proof['readers_joined']=not any(t.is_alive() for t in readers)
            require(not any(t.is_alive() for t in readers),'CHILD_READER_NOT_JOINED')

class GitGate:
    def __init__(self,cfg,hook,writer):
        self.root = chain(exact_path(cfg['candidate_root']))
        self.exe = chain(exact_path(cfg['git_executable']))
        require(self.exe.is_file(),'GIT_EXE_FILE')
        require(self.exe.stat().st_size<=16777216,'GIT_EXE_SIZE')
        self.exe_hash = digest(self.exe.read_bytes())
        require(self.exe_hash==cfg['git_executable_sha256'].lower(),'GIT_EXE_HASH')
        self.hook,self.writer,self.proofs = hook,writer,[]
        self.__verified = False
        self.__calls = 0
        self.__verify_started = False
    @property
    def verified(self):
        return self.__verified
    def _run(self,tail):
        require(type(tail) is tuple and all(type(x) is str for x in tail),'GIT_REQUEST_TYPE')
        require(tail in IDENTITIES or (tail==SOURCE_TAIL and self.__verified),'GIT_TAIL')
        require(self.__calls<5,'GIT_CALL_CAP')
        self.__calls += 1
        win = chain(exact_path(os.environ['SystemRoot']))
        env = {'SystemRoot':str(win),'WINDIR':str(win),'GIT_CONFIG_NOSYSTEM':'1',
          'GIT_CONFIG_GLOBAL':os.devnull,'GIT_TERMINAL_PROMPT':'0','GIT_OPTIONAL_LOCKS':'0',
          'PATH':str(self.exe.parent)+';'+str(win/'System32'),'LANG':'C.UTF-8','LC_ALL':'C.UTF-8'}
        argv = (str(self.exe),'--no-optional-locks','--no-pager','-C',str(self.root),*tail)
        p = {'argv':argv,'rendered':subprocess.list2cmdline(argv),'exe':str(self.exe),
          'exe_hash':self.exe_hash,'cwd':str(self.root),'env':env,'nonce':uuid.uuid4().hex,
          'audit_consumed':False,'out_cap':1048576 if tail==SOURCE_TAIL else 4096}
        proof={'tail':list(tail)}
        try:
            out,err = bounded_child(p,self.hook,proof)
        finally:
            self.proofs.append(proof)
            self.writer.json('git_'+str(len(self.proofs))+'.json',proof)
        require(not err,'CHILD_STDERR_NONEMPTY')
        return out.decode('utf-8',errors='strict').strip()
    def verify(self):
        require(not self.__verify_started,'GIT_VERIFY_RETRY_DENIED')
        self.__verify_started = True
        self.__verified = False
        self.proofs.clear()
        # Only these actual child results can advance the identity state.
        actual = [self._run(tail) for tail in IDENTITIES]
        require(actual[:3]==[HEAD,TREE,PARENT],'GIT_IDENTITY_MISMATCH')
        common = chain(exact_path(actual[3].replace('/','\\')))
        require(common==self.root/'.git','GIT_COMMON_DIR_MISMATCH')
        require(len(self.proofs)==4 and all(p['audit_consumed'] and p['reaped'] and
          p['exit_code']==0 and p['job_assigned'] and p['readers_joined'] and
          p['streams_complete'] for p in self.proofs),'GIT_IDENTITY_PROOF')
        self.__verified = True
    def source_map(self):
        require(self.__verified,'GIT_SOURCE_BEFORE_IDENTITY')
        entries = {}
        for row in self._run(SOURCE_TAIL).splitlines():
            meta,rel = row.split('\t',1)
            mode,kind,identity = meta.split()
            require(mode=='100644' and kind=='blob','SOURCE_TREE_ENTRY')
            require(rel.startswith('src/ch5_two_asset_hank/') and '..' not in rel.split('/'),
                    'SOURCE_TREE_PATH')
            entries[rel]=identity
        return entries

class VerifiedLoader(importlib.abc.Loader):
    def __init__(self,finder,fullname,rel,is_package):
        self.finder,self.fullname,self.rel,self.is_package = finder,fullname,rel,is_package
    def create_module(self,spec):
        return None
    def exec_module(self,module):
        path,raw = self.finder.disk(self.rel)
        # Compile exactly the authenticated bytes; never use or create pycache.
        module.__file__ = str(path)
        if self.is_package:
            module.__path__ = [str(path.parent)]
        self.finder.loaded[self.fullname] = {'path':str(path),'rel':self.rel,
          'git_blob':self.finder.entries[self.rel],'raw_sha256':digest(raw)}
        exec(compile(raw,str(path),'exec',dont_inherit=True),module.__dict__)

class VerifiedFinder(importlib.abc.MetaPathFinder):
    def __init__(self,gate,entries):
        require(gate.verified,'SOURCE_IDENTITY_NOT_VERIFIED')
        self.root,self.entries,self.loaded = gate.root,entries,{}
    def disk(self,rel):
        require(rel in self.entries and rel.endswith('.py'),'SOURCE_UNTRACKED')
        path = chain(self.root/Path(rel))
        require(path.is_relative_to(self.root/'src'/'ch5_two_asset_hank'),'SOURCE_OUTSIDE_ROOT')
        require(path.stat().st_size<=4194304,'SOURCE_FILE_CAP')
        raw = path.read_bytes()
        require(len(raw)<=4194304,'SOURCE_FILE_CAP')
        # Authenticate the very bytes to be compiled; no newline normalization.
        require(blob(raw)==self.entries[rel],'SOURCE_DIRTY_BYTES')
        raw.decode('utf-8',errors='strict')
        return path,raw
    def find_spec(self,fullname,path=None,target=None):
        if not (fullname=='ch5_two_asset_hank' or fullname.startswith('ch5_two_asset_hank.')):
            return None
        base = 'src/'+fullname.replace('.','/')
        for rel,pkg in ((base+'/__init__.py',True),(base+'.py',False)):
            if rel in self.entries:
                loader = VerifiedLoader(self,fullname,rel,pkg)
                return importlib.util.spec_from_loader(fullname,loader,is_package=pkg)
        raise Stop('SOURCE_UNTRACKED_IMPORT')
    def check_loaded(self):
        for suffix,wanted in NEEDED.items():
            rel = 'src/ch5_two_asset_hank/'+suffix
            require(self.entries.get(rel)==wanted,'SOURCE_NINE_TREE_BLOB')
            name = 'ch5_two_asset_hank.'+suffix[:-3].replace('/','.')
            require(name in self.loaded,'SOURCE_NINE_NOT_IMPORTED')
            path,raw = self.disk(rel)
            module = sys.modules.get(name)
